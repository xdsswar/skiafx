# ---------------------------------------------------------------------------
# glib-lite.dll — JFX-bundled GLib + libffi (Windows).
#
# Translated from upstream Makefiles:
#   gstreamer/projects/win/glib-lite/Makefile{,.ffi,.glib,.gmodule,.gobject,.gthread}
#
# Build shape: five OBJECT libraries (one per sub-Makefile) compiled
# with their respective defines + includes, then linked together into
# one SHARED library. The OBJECT-library split mirrors the upstream
# Makefile structure exactly, so any future flag drift can be ported
# without restructuring.
#
# Exports: controlled by `glib-lite.def` (in the upstream build/win32
# tree, supplied via /DEF: link flag).
# ---------------------------------------------------------------------------

# Source-tree roots. NATIVE_ROOT = modules/javafx.media/src/main/native/
set(NATIVE_ROOT  ${CMAKE_CURRENT_SOURCE_DIR}/../..)
set(GLIB_BASE    ${NATIVE_ROOT}/gstreamer/3rd_party/glib)

# Common MSVC compile options (with widely-disabled warnings). The
# upstream Makefiles disable these aggressively because the GLib code
# was originally GCC-targeted; modern MSVC flags more conformance issues
# than the GLib code is willing to address.
set(GLIB_COMMON_OPTIONS
    /nologo /W3 /WX- /EHsc /GS /fp:precise /Gm- /Zc:wchar_t /Zc:forScope /Gd
    /analyze- /errorReport:queue
    /wd4430 /wd4005 /wd4018 /wd4028 /wd4090 /wd4113 /wd4146 /wd4267
    /wd4311 /wd4312 /wd4133 /wd4334 /wd4715 /wd4116
    "/FImsvc_recommended_pragmas.h"
)
# Force-include `msvc_recommended_pragmas.h` finds it relative to the
# include path — needs the build/win32/vs100 include to be on the line.

# Release-only optimization flags (matching Makefile.glib's release
# branch); Debug build adds -Od -RTC1 -MDd via CMAKE_C_FLAGS_DEBUG.
set(GLIB_RELEASE_OPTIONS /O1 /Oy /MD /Gy /GF)

# Common defines shared by glib/gmodule/gobject/gthread (libffi has
# its own narrower set, applied per-target below).

set(GLIB_RELEASE_DEFINES NDEBUG G_DISABLE_ASSERT)

# closures.c calls ffi_tramp_is_supported/_alloc/_get_addr but those
# aren't defined anywhere in the libffi tree we have (upstream JFX's
# Makefile.ffi doesn't link tramp.c either). closures.c handles
# "tramp not supported" gracefully — every call site has an
# `if (!ffi_tramp_is_supported()) ...` fallback. So we generate a
# tiny stub source that makes that branch always taken.
set(FFI_TRAMP_STUB ${CMAKE_CURRENT_BINARY_DIR}/ffi_tramp_stub.c)
file(WRITE ${FFI_TRAMP_STUB}
"/* Auto-generated by CMakeLists.txt: trampolines are not used in
 * this static Windows build. Returning \"not supported\" lets
 * closures.c fall back to its non-tramp paths. */
int ffi_tramp_is_supported(void) { return 0; }
void *ffi_tramp_alloc(int flags) { (void)flags; return 0; }
void *ffi_tramp_get_addr(void *t) { (void)t; return 0; }
void ffi_tramp_free(void *t) { (void)t; }
")

# libffi's x64 assembly entry — preprocessed by cl /EP, then assembled
# by ml64. The upstream Makefile.ffi runs `cl -EP` to expand macros
# in `win64_intel.S` then pipes through ml64. CMake's MASM language
# doesn't run a preprocessor by default, so we add a custom command
# that does the same dance.
set(FFI_ASM_SRC ${LIBFFI_BASE}/src/x86/win64_intel.S)
set(FFI_ASM_PRE ${CMAKE_CURRENT_BINARY_DIR}/win64_intel.preprocessed.asm)
set(FFI_ASM_OBJ ${CMAKE_CURRENT_BINARY_DIR}/win64_intel.asm.obj)

# Pre-process the .S with cl /EP /Fi so #defines from the libffi
# headers expand (the file uses #include and #define heavily). /Fi
# names the output file directly — no shell redirection needed.
# cmd /c is required because cl exits with non-zero on /EP /Fi
# combinations in some toolchains; wrapping in cmd /c lets us mask
# that with `& exit 0` after verifying the output exists. Simpler:
# capture stdout to file via cmd /c "...".
add_custom_command(
    OUTPUT ${FFI_ASM_PRE}
    COMMAND cmd /c cl /EP /DFFI_STATIC_BUILD /DX86_WIN64
            /I "${LIBFFI_BASE}/include"
            /I "${LIBFFI_BASE}/include/win/x64"
            /I "${LIBFFI_BASE}/src/x86"
            "${FFI_ASM_SRC}" > "${FFI_ASM_PRE}"
    DEPENDS ${FFI_ASM_SRC}
)
# Assemble the preprocessed output with ml64. ml64's /Fo arg doesn't
# accept forward slashes in the output path on some toolchain versions
# — feed native (backslash) paths through file(TO_NATIVE_PATH).
file(TO_NATIVE_PATH "${FFI_ASM_OBJ}" FFI_ASM_OBJ_NATIVE)
file(TO_NATIVE_PATH "${FFI_ASM_PRE}" FFI_ASM_PRE_NATIVE)
add_custom_command(
    OUTPUT ${FFI_ASM_OBJ}
    COMMAND ml64 /nologo /c /Fo${FFI_ASM_OBJ_NATIVE} ${FFI_ASM_PRE_NATIVE}
    DEPENDS ${FFI_ASM_PRE}
)
# Wrap the .obj in a custom target so the final glib-lite link picks it up.
add_custom_target(glib-lite-ffi-asm DEPENDS ${FFI_ASM_OBJ})

    $<TARGET_OBJECTS:glib-lite-glib>
    $<TARGET_OBJECTS:glib-lite-gmodule>
    $<TARGET_OBJECTS:glib-lite-gobject>

)
# The asm .obj is produced by a custom_command; mark it explicitly
# so CMake adds it to the link line as a plain object file.
set_source_files_properties(${FFI_ASM_OBJ} PROPERTIES
    EXTERNAL_OBJECT TRUE
    GENERATED TRUE
)
add_dependencies(glib-lite glib-lite-ffi-asm)

# Drop the default "lib" prefix MSVC otherwise omits, and force the
# DLL name to match what gstreamer-lite later wants to link against
# (-l glib-lite ⇒ glib-lite.lib import library + glib-lite.dll).
set_target_properties(glib-lite PROPERTIES
    OUTPUT_NAME glib-lite
    PREFIX ""
)

# Exports via .def file.
target_link_options(glib-lite PRIVATE
    /DEF:${GLIB_DEF}
    /SUBSYSTEM:WINDOWS
    /TLBID:1
    /DYNAMICBASE
    /NXCOMPAT
    /ERRORREPORT:QUEUE
    /MANIFESTUAC:"level='asInvoker' uiAccess='false'"
    $<$<CONFIG:Release>:/OPT:REF /OPT:ICF>
)

target_link_libraries(glib-lite PRIVATE
    Ws2_32.lib
    kernel32.lib
    user32.lib
    shell32.lib
    advapi32.lib

    Winmm.lib
)
